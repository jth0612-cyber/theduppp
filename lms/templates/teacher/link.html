시트에 붙일 코드 (WebSync.gs) 
// THedu 통합과학 lab 웹으로 앱용 표 보내기
var WEB_URL = '{{ base }}';
var APP_SHEETS = ['🌐앱_성적', '🌐앱_오답노트', '🌐앱_리포트', '🌐앱_과제', '🌐앱_출결', '🌐앱_공부유형', '🌐앱_학습유형'];

function pushToWeb() {
  var ss = SpreadsheetApp.getActive(), tables = {};
  APP_SHEETS.forEach(function (name) {
    var sh = ss.getSheetByName(name);
    if (sh && sh.getLastRow() > 0) tables[name] = sh.getDataRange().getDisplayValues();
  });
  // 가입 코드용: 학생코드·반·상태만 (이름·연락처는 보내지 않는다)
  var db = ss.getSheetByName('학생DB');
  if (db) {
    var v = db.getDataRange().getDisplayValues(), h = v[0];
    var ic = h.indexOf('학생코드'), ib = h.indexOf('반'), is = h.indexOf('상태');
    tables['코드'] = [['학생코드', '반', '상태']].concat(v.slice(1).filter(function (r) { return r[ic]; })
      .map(function (r) { return [r[ic], r[ib], is >= 0 ? r[is] : '']; }));
  }
  var res = UrlFetchApp.fetch(WEB_URL + '/api/sync', {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    headers: { Authorization: 'Bearer ' + PropertiesService.getScriptProperties().getProperty('WEB_SYNC_TOKEN') },
    payload: JSON.stringify({ tables: tables })
  });
  Logger.log(res.getResponseCode() + ' ' + res.getContentText());
}

// 학생이 웹에서 표시한 '다시 풀기' 결과를 '🌐웹_다시풀기' 시트로 가져온다
function pullRedoFromWeb() {
  var res = UrlFetchApp.fetch(WEB_URL + '/api/sync/redo', { muteHttpExceptions: true,
    headers: { Authorization: 'Bearer ' + PropertiesService.getScriptProperties().getProperty('WEB_SYNC_TOKEN') } });
  var rows = JSON.parse(res.getContentText()).rows || [];
  var sh = SpreadsheetApp.getActive().getSheetByName('🌐웹_다시풀기') || SpreadsheetApp.getActive().insertSheet('🌐웹_다시풀기');
  sh.clearContents(); if (rows.length) sh.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
}
